cmake_minimum_required(VERSION 3.16)
project(StreamMatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(recommender src/recommender.cpp)
target_include_directories(recommender PUBLIC src)
add_executable(streammatch src/main.cpp)
target_link_libraries(streammatch PRIVATE recommender)
foreach(target recommender streammatch)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
enable_testing()
add_executable(unit_tests tests/test_recommender.cpp)
target_link_libraries(unit_tests PRIVATE recommender)
if(MSVC)
    target_compile_options(unit_tests PRIVATE /W4 /WX)
else()
    target_compile_options(unit_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME unit_tests COMMAND unit_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME cli_tests COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:streammatch>)
endif()
